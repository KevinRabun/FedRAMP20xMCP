## Bicep Template

```bicep
// evidence_minimal.bicep - Minimal evidence storage
// Smallest hardened deployment: one evidence account and one immutable container.
//
// CR26 (FedRAMP Consolidated Rules for 2026) references for evidence storage:
//   KSI-MLA-RVL / KSI-MLA-OSM / KSI-MLA-ALA  log review, SIEM/log aggregation, log access
//   KSI-SVC-SIN  information secured       KSI-SVC-RUD  removing unwanted data
//   KSI-CMT-LMC  logging changes           VER-TFR-MRH  historical activity
//   IVV-CSO-SEI / IVV-CSO-SEE  supply evidence of implementation / effectiveness
//   FRC-CSX-VVK / FRC-CSX-VVR  automated verification and validation (KSIs / rules)
//   SDR-CSX-KMT  KSI metrics
// CR26 sets no numeric log or evidence retention period. The 1095 day (3 year) default is an
// organizational choice; keep it configurable.
//
// Blob layout convention (one container per evidence stream):
//   <container>/<report-or-app>/<yyyy-MM-dd or version>/<runId or buildId>/<files>
//   Each run folder also carries:
//     run.json             provenance: pipeline, buildId, commit, rulesVersion, collectedAt
//     sha256-manifest.txt  SHA-256 of every evidence file in the folder
//   Upload with Entra auth and WITHOUT --overwrite (versioning is off on purpose; with
//   container-level WORM an overwrite of an existing blob name fails instead of silently
//   creating a version):
//     az storage blob upload-batch --auth-mode login --account-name <account> \
//       --destination <container> --destination-path <report-or-app>/<date>/<runId> --source ./out
//
// Locking is a deliberate, one-way, out-of-band step. Once a time-based retention policy is
// locked it can only be extended (never shortened or removed) and the storage account cannot
// be deleted while retained blobs exist. Review the unlocked policy first, then lock it with
// `az storage container immutability-policy lock` (or the portal).
// Locking is intentionally not a deployment parameter (the Bicep resource cannot lock it).
//
// Lifecycle: tier to cool after 30 days since creation. There is deliberately NO delete
// rule. Expiry of retained evidence is a records-management decision, not a lifecycle default.
//
// Baseline: StorageV2 Standard RA-GRS (parameterized), Hot tier, shared key and SAS disabled,
// public network access disabled, private endpoint (blob) with private DNS zone group,
// optional customer-managed key (user-assigned identity + versionless Key Vault key),
// infrastructure encryption, no SFTP/NFSv3, versioning OFF, change feed ON, no blob soft
// delete or point-in-time restore (incompatible with immutability), container soft delete 90d.
// Writers: one user-assigned managed identity per stream with Storage Blob Data Contributor
// scoped to its own container. Readers: an Entra group with Storage Blob Data Reader on the
// account. No access keys, SAS tokens or connection strings with AccountKey anywhere.

targetScope = 'resourceGroup'

@description('Deployment location')
param location string = resourceGroup().location

@description('Unique suffix for resource naming')
param nameSuffix string = uniqueString(resourceGroup().id)

@description('Storage replication. RA-GRS by default (organizational baseline).')
@allowed([
  'Standard_RAGRS'
  'Standard_RAGZRS'
  'Standard_GRS'
  'Standard_GZRS'
  'Standard_ZRS'
  'Standard_LRS'
])
param storageReplication string = 'Standard_RAGRS'

@description('Container immutability (WORM) period in days since creation. 1095 (3 years) is an organizational choice, not a FedRAMP requirement.')
@minValue(1)
param immutabilityPeriodDays int = 1095

@description('Use a customer-managed key when the key inputs below are provided.')
param enableCmk bool = true

@description('Key Vault URI holding the CMK, e.g. https://kv-name.vault.azure.net/ (empty = Microsoft-managed key)')
param cmkKeyVaultUri string = ''

@description('Key name in the Key Vault. The key version is omitted so rotation is automatic.')
param cmkKeyName string = ''

@description('Resource ID of the user-assigned identity with Key Vault Crypto Service Encryption User on the key (empty = Microsoft-managed key)')
param cmkIdentityId string = ''

@description('Subnet resource ID for the blob private endpoint')
param privateEndpointSubnetId string

@description('Resource ID of the privatelink.blob.core.windows.net private DNS zone')
param blobPrivateDnsZoneId string

@description('Log Analytics workspace resource ID for StorageRead/StorageWrite/StorageDelete diagnostics')
param logAnalyticsWorkspaceId string

@description('Object ID of the Entra group granted Storage Blob Data Reader on the account (use a group, not individuals)')
param readerGroupObjectId string

@description('Evidence container name (one stream)')
param containerName string = 'evidence-reports'

var cmkEnabled = enableCmk && !empty(cmkKeyVaultUri) && !empty(cmkKeyName) && !empty(cmkIdentityId)
var storageBlobDataContributorRoleId = 'ba92f5b4-2d11-453d-a403-e96b0029c9fe'
var storageBlobDataReaderRoleId = '2a2b9908-6ea1-4ae2-8e65-a410df84e7d1'
var streams = [
  containerName
]

var tags = {
  Compliance: 'FedRAMP 20x CR26'
  Purpose: 'Evidence Storage'
  ImmutabilityDays: string(immutabilityPeriodDays)
}

// ============================================================================
// Evidence Storage Account
// Supports: KSI-SVC-SIN (information secured), KSI-SVC-RUD (removing unwanted data)
// ============================================================================

resource storageAccount 'Microsoft.Storage/storageAccounts@2023-05-01' = {
  name: 'stevidmin${take(nameSuffix, 12)}'
  location: location
  tags: tags
  sku: {
    name: storageReplication
  }
  kind: 'StorageV2'
  identity: cmkEnabled ? {
    type: 'UserAssigned'
    userAssignedIdentities: {
      '${cmkIdentityId}': {}
    }
  } : null
  properties: {
    accessTier: 'Hot'
    allowSharedKeyAccess: false // Entra only: no account keys, no key-signed SAS
    defaultToOAuthAuthentication: true
    allowBlobPublicAccess: false
    publicNetworkAccess: 'Disabled'
    supportsHttpsTrafficOnly: true
    minimumTlsVersion: 'TLS1_2'
    isLocalUserEnabled: false
    isSftpEnabled: false // SFTP and NFSv3 are incompatible with immutability
    isNfsV3Enabled: false
    allowCrossTenantReplication: false
    allowedCopyScope: 'AAD'
    networkAcls: {
      defaultAction: 'Deny'
      bypass: 'AzureServices'
    }
    encryption: {
      keySource: cmkEnabled ? 'Microsoft.Keyvault' : 'Microsoft.Storage'
      requireInfrastructureEncryption: true
      services: {
        blob: {
          enabled: true
          keyType: 'Account'
        }
        file: {
          enabled: true
          keyType: 'Account'
        }
      }
      identity: cmkEnabled ? {
        userAssignedIdentity: cmkIdentityId
      } : null
      keyvaultproperties: cmkEnabled ? {
        keyname: cmkKeyName
        keyvaulturi: cmkKeyVaultUri // keyversion omitted: versionless key, rotation is automatic
      } : null
    }
  }
}

resource blobService 'Microsoft.Storage/storageAccounts/blobServices@2023-05-01' = {
  parent: storageAccount
  name: 'default'
  properties: {
    // Versioning is OFF on purpose: with container-level WORM and no versioning, an
    // overwrite of an existing blob name fails outright instead of silently creating a version.
    isVersioningEnabled: false
    changeFeed: {
      enabled: true // audit trail of blob changes
    }
    containerDeleteRetentionPolicy: {
      enabled: true
      days: 90
    }
    // No blob soft delete and no point-in-time restore: incompatible with immutability.
    deleteRetentionPolicy: {
      enabled: false
    }
    restorePolicy: {
      enabled: false
    }
  }
}

// ============================================================================
// Containers (one per evidence stream), each private and WORM
// ============================================================================

resource evidenceContainers 'Microsoft.Storage/storageAccounts/blobServices/containers@2023-05-01' = [for stream in streams: {
  parent: blobService
  name: stream
  properties: {
    publicAccess: 'None'
  }
}]

// Time-based retention. Created UNLOCKED. See the locking note at the top of this file.
resource immutabilityPolicies 'Microsoft.Storage/storageAccounts/blobServices/containers/immutabilityPolicies@2023-05-01' = [for (stream, i) in streams: {
  parent: evidenceContainers[i]
  name: 'default'
  properties: {
    immutabilityPeriodSinceCreationInDays: immutabilityPeriodDays
    allowProtectedAppendWrites: false
  }
}]

// ============================================================================
// Lifecycle management (tier only, never delete)
// ============================================================================

resource lifecyclePolicy 'Microsoft.Storage/storageAccounts/managementPolicies@2023-05-01' = {
  parent: storageAccount
  name: 'default'
  properties: {
    policy: {
      rules: [
        {
          name: 'tier-evidence-to-cool-30d'
          enabled: true
          type: 'Lifecycle'
          definition: {
            filters: {
              blobTypes: [
                'blockBlob'
              ]
            }
            actions: {
              baseBlob: {
                tierToCool: {
                  daysAfterCreationGreaterThan: 30
                }
                // No delete action: retention expiry is a records decision.
              }
            }
          }
        }
      ]
    }
  }
}

// Private endpoint (blob) with private DNS zone group
// ============================================================================

resource blobPrivateEndpoint 'Microsoft.Network/privateEndpoints@2023-09-01' = {
  name: 'pe-${storageAccount.name}-blob'
  location: location
  tags: tags
  properties: {
    subnet: {
      id: privateEndpointSubnetId
    }
    privateLinkServiceConnections: [
      {
        name: 'blob'
        properties: {
          privateLinkServiceId: storageAccount.id
          groupIds: [
            'blob'
          ]
        }
      }
    ]
  }
}

resource blobPrivateDnsZoneGroup 'Microsoft.Network/privateEndpoints/privateDnsZoneGroups@2023-09-01' = {
  parent: blobPrivateEndpoint
  name: 'default'
  properties: {
    privateDnsZoneConfigs: [
      {
        name: 'privatelink-blob-core-windows-net'
        properties: {
          privateDnsZoneId: blobPrivateDnsZoneId
        }
      }
    ]
  }
}

// ============================================================================
// RBAC: per-container writers, group readers. No keys, no SAS.
// Any function, app or collector writes using one of these managed identities (Entra only).
// ============================================================================

resource writerIdentities 'Microsoft.ManagedIdentity/userAssignedIdentities@2023-01-31' = [for stream in streams: {
  name: 'id-evidence-writer-${stream}-${nameSuffix}'
  location: location
  tags: tags
}]

resource writerRoleAssignments 'Microsoft.Authorization/roleAssignments@2022-04-01' = [for (stream, i) in streams: {
  scope: evidenceContainers[i] // container scope, not account: a writer cannot touch another stream
  name: guid(evidenceContainers[i].id, writerIdentities[i].id, storageBlobDataContributorRoleId)
  properties: {
    roleDefinitionId: subscriptionResourceId('Microsoft.Authorization/roleDefinitions', storageBlobDataContributorRoleId)
    principalId: writerIdentities[i].properties.principalId
    principalType: 'ServicePrincipal'
  }
}]

resource readerRoleAssignment 'Microsoft.Authorization/roleAssignments@2022-04-01' = {
  scope: storageAccount
  name: guid(storageAccount.id, readerGroupObjectId, storageBlobDataReaderRoleId)
  properties: {
    roleDefinitionId: subscriptionResourceId('Microsoft.Authorization/roleDefinitions', storageBlobDataReaderRoleId)
    principalId: readerGroupObjectId
    principalType: 'Group'
  }
}

// Diagnostics: StorageRead / StorageWrite / StorageDelete to Log Analytics
// Supports: KSI-MLA-OSM, KSI-MLA-ALA, KSI-CMT-LMC
// ============================================================================

resource blobDiagnostics 'Microsoft.Insights/diagnosticSettings@2021-05-01-preview' = {
  name: 'evidence-blob-to-law'
  scope: blobService
  properties: {
    workspaceId: logAnalyticsWorkspaceId
    logs: [
      {
        category: 'StorageRead'
        enabled: true
      }
      {
        category: 'StorageWrite'
        enabled: true
      }
      {
        category: 'StorageDelete'
        enabled: true
      }

    metrics: [
      {
        category: 'Transaction'
        enabled: true
      }
    ]
  }
}

output storageAccountName string = storageAccount.name
output storageAccountId string = storageAccount.id
output containerNames array = streams
output writerIdentities array = [for (stream, i) in streams: {

  clientId: writerIdentities[i].properties.clientId
```

## Usage Notes

- Public network access is disabled: upload from a runner or agent with private connectivity to the endpoint.
- Writers authenticate with their managed identity or federated workload identity (Entra). Shared key access is disabled, so connection strings with AccountKey and SAS tokens do not work.
- The immutability policies are created unlocked. Lock them out of band when ready (see the locking note in the template).
- Retention (1095 days) is an organizational choice, not a FedRAMP number. CR26 does not set one.
- Grant the CMK identity Key Vault Crypto Service Encryption User on the key before deploying with CMK. Omit the cmk parameters to use a Microsoft-managed key.
