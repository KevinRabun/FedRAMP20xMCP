````plaintext
// frr_ads.bicep - Certification Data Sharing API Infrastructure
// (The former ADS family is now CDS in CR26; the file name is kept for caller compatibility.)
// Infrastructure sketch for sharing FedRAMP Certification Data with programmatic access,
// authentication, access logging and automated evidence collection.
//
// CDS rules supported:
// - CDS-CSO-PUB: Public information in human-readable and JSON formats
// - CDS-CSO-UTC: Use a FedRAMP-compatible trust center
// - CDS-CSO-CBF: Automation keeps human-readable and machine-readable formats consistent
// - CDS-TRC-PAC: Documented programmatic access to all Certification Data
// - CDS-TRC-ACL: Log access and keep access summaries for at least six months
// - CDS-TRC-AAI: Inventory and history of agency users with access
//
// Related KSIs:
// - KSI-IAM-APM: Authentication
// - KSI-CNA-RNT: Network security controls
// - KSI-MLA-OSM: Logging
//
// Container and endpoint names are design suggestions, not CR26 requirements.

targetScope = 'resourceGroup'

param location string = resourceGroup().location
param apiManagementName string = 'apim-cds-${uniqueString(resourceGroup().id)}'
param functionAppName string = 'func-cds-${uniqueString(resourceGroup().id)}'
param storageAccountName string = 'cdsevidence${uniqueString(resourceGroup().id)}'
param keyVaultName string = 'kv-cds-${uniqueString(resourceGroup().id)}'
@description('Log retention in days. Organization-defined: CR26 sets no numeric log retention period (CDS-TRC-ACL separately requires access summaries for at least six months).')
param logRetentionDays int

// Key Vault for API keys and secrets
// Supports: CDS-CSO-UTC (access control), KSI-SVC-ASM (secret management), CMU-CSO-CMD (cryptographic module documentation)
resource keyVault 'Microsoft.KeyVault/vaults@2023-07-01' = {
  name: keyVaultName
  location: location
  properties: {
    sku: {
      family: 'A'
      name: 'premium'  // HSM-backed keys; choose the SKU per CMU-CSO-UVM for your Certification Class
    }
    tenantId: subscription().tenantId
    enableRbacAuthorization: true
    enablePurgeProtection: true
    enableSoftDelete: true
    softDeleteRetentionInDays: 90
    networkAcls: {
      defaultAction: 'Deny'
      bypass: 'AzureServices'
      ipRules: []
      virtualNetworkRules: []
    }
    publicNetworkAccess: 'Disabled'  // KSI-CNA-RNT: private access
  }
}

      virtualNetworkRules: []
      ipRules: []

    encryption: {
      requireInfrastructureEncryption: true
      services: {
        blob: {
          enabled: true
          keyType: 'Account'
        }
        file: {
          enabled: true
          keyType: 'Account'
        }
      }
      keySource: 'Microsoft.Keyvault'  // KSI-SVC-ASM: customer-managed keys (design default)
      keyvaultproperties: {
        keyname: 'storage-encryption-key'
        keyvaulturi: keyVault.properties.vaultUri
      }
    }
  }
}

## Deployment Instructions

1. **Create Resource Group:**
   ```bash
   az group create --name rg-cds-compliance --location eastus
   ```

2. **Deploy Template:**
   ```bash
   az deployment group create \
     --resource-group rg-cds-compliance \
     --template-file frr_ads.bicep \
     --parameters logRetentionDays=<organization-defined>
   ```

3. **Deploy Azure Functions Code:**
   ```bash
   cd azure-functions/cds-api
   func azure functionapp publish func-cds-<unique-id>
   ```

4. **Configure API Management Policies:**
   - Add OAuth2 authentication (Entra ID)
   - Configure rate limiting suited to your consumers (design choice)
   - Enable response caching for static data
   - Set up subscription key management

5. **Test API Endpoints:**
   ```bash
   curl -H "Ocp-Apim-Subscription-Key: YOUR_KEY" \
     https://apim-cds-<unique-id>.azure-api.net/certification-data/security-decision-record
   ```

## Suggested API Endpoints (design suggestion, not CR26 requirements)

| Endpoint | Rules | Description |
|----------|-------|-------------|
| `/ongoing-certification-report` | CCM-OCR-AVL | Ongoing Certification Report data |
| `/security-decision-record` | SDR-CSO-FRR, SDR-CSX-KSI | Per-rule and per-KSI explanations, verification, validation |
| `/vulnerabilities` | VER-RPT-PER, VER-RPT-VDT, VER-RPT-AVI | Vulnerability reports, accepted vulnerabilities |
| `/incidents` | IEC-CSO-IIR, IEC-CSO-OIR, IEC-CSO-FIR | Incident reports |
| `/significant-changes` | SCN-CSO-HIS, SCN-CSO-HRM | Significant Change Notifications (12 months) |
| `/access-log-summary` | CDS-TRC-ACL, CDS-TRC-AAI | Access log summaries and agency access inventory |

## Machine-Readable Formats (CDS-CSO-PUB)

All endpoints return JSON with the following structure:

{
  "metadata": {

    "version": "1.0",
    "classification": "Controlled Unclassified Information"
  },
  "data": {
    // Compliance data specific to endpoint
  }
}
```

## Authentication (CDS-TRC-PAC, KSI-IAM-APM)

- **Method**: OAuth 2.0 (Microsoft Entra ID)
- **Scopes**: `api://cds-api/read` (use write scopes only for the publishing pipeline)
- **Subscription Keys**: Required for all API calls
- **IP Allowlist**: Configure in API Management policies

## Evidence Collection Schedule

Choose cadences from the rules that apply to your Certification Class, for example:
- **Vulnerability reports**: at least monthly (VER-TFR-MHR)
- **Ongoing Certification Report**: every 3 months (CCM-OCR-AVL)
- **Certification Package maintenance**: CPO-CSX-CPM (20x) or CPO-CSF-CPM (Rev5), by Class
- **Certification Data snapshots**: aligned to each Ongoing Certification Report (CDS-CSO-HAD)
````
