/*
 * CDS Certification Data Sharing API (C#)
 *
 * This module sketches a trust-center style API for FedRAMP Certification Data
 * under the FedRAMP Consolidated Rules for 2026 (CR26). The former ADS family
 * (Authorization Data Sharing) is now the CDS family (Certification Data Sharing).
 * The template file name (frr_ads) is kept for caller compatibility.
 *
 * CDS Rules Supported:
 * - CDS-CSO-PUB: Public information in human-readable and JSON formats
 * - CDS-CSO-CBF: Automation keeps human-readable and machine-readable formats consistent
 * - CDS-CSO-UTC: Use a FedRAMP-compatible trust center
 * - CDS-TRC-PAC: Documented programmatic access to all Certification Data
 * - CDS-TRC-ACL: Log access to Certification Data and keep access summaries for at least six months
 * - CDS-TRC-AAI: Inventory and history of federal agency users with access
 *
 * Related reporting rules:
 * - VER-RPT-PER, VER-RPT-VDT, VER-RPT-AVI: Vulnerability reports (Certification Data)
 * - VDR-TFR-KEV: Known Exploited Vulnerabilities
 * - CCM-OCR-AVL: Ongoing Certification Report
 * - SDR-CSO-FRR, SDR-CSX-KSI: Security Decision Record
 *
 * Related KSIs:
 * - KSI-IAM-APM: Authentication
 * - KSI-CNA-RNT: Network security controls
 *
 * The endpoint layout is a design suggestion, not a CR26 requirement. JSON served
 * publicly should validate against the FedRAMP JSON schema where a rule defines one
 * (FRC-CSO-JSN).
 *
 * Usage:
 *   dotnet run --project CdsApi
 *   curl -H "Authorization: Bearer TOKEN" https://api.contoso.com/api/certification-data/security-decision-record
 */

using System;
using System.Collections.Generic;
using System.Linq;
using System.Threading.Tasks;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Mvc;
using Microsoft.Extensions.Logging;
using Azure.Identity;
using Azure.Storage.Blobs;
using Azure.Security.KeyVault.Secrets;
using Azure.Monitor.Query;

namespace CdsApi.Controllers
{
    /// <summary>
    /// Certification Data Sharing API Controller
    ///
    /// Provides programmatic access to FedRAMP Certification Data
    /// per CDS-TRC-PAC and the related reporting rules.
    /// </summary>
    [ApiController]
    [Route("api/certification-data")]
    [Authorize] // CDS-TRC-PAC / KSI-IAM-APM: authenticated access for non-public data
    public class CertificationDataController : ControllerBase
    {
        private readonly ILogger<CertificationDataController> _logger;
        private readonly BlobServiceClient _blobClient;
        private readonly SecretClient _keyVaultClient;
        private readonly LogsQueryClient _logsClient;

        // Data categories and the rules they support
        private static readonly Dictionary<string, string> DataCategories = new()
        {
            { "security-decision-record", "SDR-CSO-FRR, SDR-CSX-KSI" },
            { "vulnerabilities", "VER-RPT-PER, VER-RPT-VDT" },
            { "ongoing-certification-report", "CCM-OCR-AVL" },
            { "significant-changes", "SCN-CSO-HIS, SCN-CSO-HRM" },
            { "access-log-summary", "CDS-TRC-ACL" },
            { "posture-summary", "CDS-CSO-PUB" }
        };

        public CertificationDataController(
            ILogger<CertificationDataController> logger,
            IConfiguration configuration)
        {
            _logger = logger;

            // Managed identity: no credentials in code (KSI-IAM-SNU)
            var credential = new DefaultAzureCredential();

            var storageAccountName = configuration["EVIDENCE_STORAGE_ACCOUNT"];
            _blobClient = new BlobServiceClient(
                new Uri($"https://{storageAccountName}.blob.core.windows.net"),
                credential
            );

            var keyVaultUri = configuration["KEY_VAULT_URI"];
            _keyVaultClient = new SecretClient(new Uri(keyVaultUri), credential);

            _logsClient = new LogsQueryClient(credential);

            _logger.LogInformation("CDS API Controller initialized");
        }

        /// <summary>
        /// Get Security Decision Record data (SDR-CSO-FRR, SDR-CSX-KSI)
        ///
        /// Returns per-rule and per-KSI explanations, verification and validation.
        /// </summary>
        /// <returns>Security Decision Record data in JSON format</returns>
        [HttpGet("security-decision-record")]
        [ProducesResponseType(typeof(CertificationDataResponse<SecurityDecisionEntry>), StatusCodes.Status200OK)]
        [ProducesResponseType(StatusCodes.Status401Unauthorized)]
        [ProducesResponseType(StatusCodes.Status500InternalServerError)]
        public async Task<IActionResult> GetSecurityDecisionRecord()
        {
            try
            {
                _logger.LogInformation("SDR-CSO-FRR: Retrieving Security Decision Record data");

                var containerClient = _blobClient.GetBlobContainerClient("security-controls");
                var blobClient = containerClient.GetBlobClient("latest-security-decision-record.json");

                var response = await blobClient.DownloadContentAsync();
                var entries = response.Value.Content.ToObjectFromJson<List<SecurityDecisionEntry>>();

                var result = new CertificationDataResponse<SecurityDecisionEntry>
                {
                    Metadata = new CertificationMetadata
                    {
                        Timestamp = DateTime.UtcNow,
                        Version = "1.0",
                        Rules = "SDR-CSO-FRR, SDR-CSX-KSI",
                        Classification = "Controlled Unclassified Information"
                    },
                    Data = entries
                };

                _logger.LogInformation($"Retrieved {entries.Count} Security Decision Record entries");
                return Ok(result);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error retrieving Security Decision Record data");
                return StatusCode(500, new { error = "Internal server error" });
            }
        }

        /// <summary>
        /// Get vulnerability detection and response data (VER-RPT-PER, VER-RPT-VDT)
        ///
        /// Fields follow VER-RPT-VDT (tracking identifier, detection and evaluation times,
        /// internet-reachable, likely exploitable, Potential Agency Impact N-rating, overdue status,
        /// final disposition). Accepted vulnerabilities (VER-TFR-MAV) use the VER-RPT-AVI fields.
        /// </summary>
        /// <param name="nRating">Optional Potential Agency Impact N-rating filter (1 to 5)</param>
        /// <param name="kevOnly">Return only Known Exploited Vulnerabilities (VDR-TFR-KEV)</param>
        /// <returns>Vulnerability data in JSON format</returns>
        [HttpGet("vulnerabilities")]
        [ProducesResponseType(typeof(CertificationDataResponse<VulnerabilityData>), StatusCodes.Status200OK)]
        public async Task<IActionResult> GetVulnerabilityData(
            [FromQuery] int? nRating = null,
            [FromQuery] bool kevOnly = false)
        {
            try
            {
                _logger.LogInformation("VER-RPT-PER: Retrieving vulnerability data");

                var containerClient = _blobClient.GetBlobContainerClient("vulnerability-data");

                var blobs = containerClient.GetBlobsAsync();
                var latestBlob = await blobs
                    .OrderByDescending(b => b.Properties.CreatedOn)
                    .FirstOrDefaultAsync();

                if (latestBlob == null)
                {
                    return NotFound(new { error = "No vulnerability data available" });
                }

                var blobClient = containerClient.GetBlobClient(latestBlob.Name);
                var response = await blobClient.DownloadContentAsync();
                var vulnData = response.Value.Content.ToObjectFromJson<VulnerabilityScanResult>();

                var filteredVulns = vulnData.Vulnerabilities.AsEnumerable();

                if (nRating.HasValue)
                {
                    filteredVulns = filteredVulns.Where(v => v.PotentialAgencyImpactNRating == nRating.Value);
                }

                if (kevOnly)
                {
                    filteredVulns = filteredVulns.Where(v => v.IsKev);
                }

                var result = new CertificationDataResponse<VulnerabilityData>
                {
                    Metadata = new CertificationMetadata
                    {
                        Timestamp = DateTime.UtcNow,
                        Version = "1.0",
                        Rules = "VER-RPT-PER, VER-RPT-VDT, VER-TFR-MHR",
                        Classification = "Controlled Unclassified Information",
                        AdditionalInfo = new Dictionary<string, object>
                        {
                            { "report_date", vulnData.Metadata.Timestamp },
                            { "total_vulnerabilities", vulnData.Vulnerabilities.Count },
                            { "filtered_count", filteredVulns.Count() },
                            { "kev_count", vulnData.Vulnerabilities.Count(v => v.IsKev) },
                            { "accepted_count", vulnData.Vulnerabilities.Count(v => v.IsAccepted) }
                        }
                    },
                    Data = filteredVulns.ToList()
                };

                _logger.LogInformation($"Retrieved {filteredVulns.Count()} vulnerabilities");
                return Ok(result);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error retrieving vulnerability data");
                return StatusCode(500, new { error = "Internal server error" });
            }
        }

        /// <summary>
        /// Get Ongoing Certification Report inputs (CCM-OCR-AVL)
        ///
        /// Returns operational metrics that feed the Ongoing Certification Report, which covers
        /// the entire period since the previous report (every 3 months under CCM-OCR-AVL).
        /// </summary>
        /// <param name="timeRange">Time range in hours (default: 24)</param>
        /// <returns>Monitoring data in JSON format</returns>
        [HttpGet("ongoing-certification-report")]
        [ProducesResponseType(typeof(CertificationDataResponse<MonitoringMetric>), StatusCodes.Status200OK)]
        public async Task<IActionResult> GetOngoingCertificationInputs([FromQuery] int timeRange = 24)
        {
            try
            {
                _logger.LogInformation("CCM-OCR-AVL: Retrieving Ongoing Certification Report inputs");

                var workspaceId = await _keyVaultClient.GetSecretAsync("log-analytics-workspace-id");

                var query = $@"
                    SecurityAlert
                    | where TimeGenerated > ago({timeRange}h)
                    | summarize
                        AlertCount = count(),
                        HighCount = countif(AlertSeverity == 'High'),
                        MediumCount = countif(AlertSeverity == 'Medium'),
                        ResolvedCount = countif(Status == 'Resolved')
                    by bin(TimeGenerated, 1h)
                    | project
                        Timestamp = TimeGenerated,
                        AlertCount,
                        HighCount,
                        MediumCount,
                        ResolvedCount
                    | order by Timestamp desc
                ";

                var queryResult = await _logsClient.QueryWorkspaceAsync(
                    workspaceId.Value.Value,
                    query,
                    new QueryTimeRange(TimeSpan.FromHours(timeRange))
                );

                var metrics = new List<MonitoringMetric>();

                foreach (var row in queryResult.Value.Table.Rows)
                {
                    metrics.Add(new MonitoringMetric
                    {
                        Timestamp = row.GetDateTimeOffset(0),
                        MetricName = "SecurityAlerts",
                        Value = row.GetInt32(1),
                        AdditionalData = new Dictionary<string, object>
                        {
                            { "high_count", row.GetInt32(2) },
                            { "medium_count", row.GetInt32(3) },
                            { "resolved_count", row.GetInt32(4) }
                        }
                    });
                }

                var result = new CertificationDataResponse<MonitoringMetric>
                {
                    Metadata = new CertificationMetadata
                    {
                        Timestamp = DateTime.UtcNow,
                        Version = "1.0",
                        Rules = "CCM-OCR-AVL",
                        Classification = "Controlled Unclassified Information"
                    },
                    Data = metrics
                };

                _logger.LogInformation($"Retrieved {metrics.Count} monitoring metrics");
                return Ok(result);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error retrieving Ongoing Certification Report inputs");
                return StatusCode(500, new { error = "Internal server error" });
            }
        }

        /// <summary>
        /// Get a summary of certification posture
        ///
        /// Aggregates data from the other endpoints. The values below are placeholders:
        /// compute them from your Security Decision Record and vulnerability data.
        /// </summary>
        /// <returns>Posture summary in JSON format</returns>
        [HttpGet("posture-summary")]
        [ProducesResponseType(typeof(CertificationDataResponse<CertificationPosture>), StatusCodes.Status200OK)]
        public IActionResult GetPostureSummary()
        {
            try
            {
                _logger.LogInformation("CDS-CSO-PUB: Retrieving posture summary");

                var posture = new CertificationPosture
                {
                    LastUpdated = DateTime.UtcNow,
                    KsiMeasuresReported = 0,   // placeholder: count from the Security Decision Record
                    KsiTotal = 46,             // CR26 defines 46 KSIs
                    VulnerabilitiesOpen = 0,   // placeholder
                    VulnerabilitiesAccepted = 0, // placeholder (VER-TFR-MAV)
                    VulnerabilitiesOverdue = 0,  // placeholder
                    ReportableIncidentsOpen = 0, // placeholder (IEC family)
                    RuleCoverage = new Dictionary<string, decimal>
                    {
                        // placeholder: share of applicable rules with explanation, verification and validation recorded
                        { "VDR", 0m },
                        { "VER", 0m },
                        { "CDS", 0m },
                        { "CCM", 0m },
                        { "SCG", 0m },
                        { "CMU", 0m }
                    }
                };

                var result = new CertificationDataResponse<CertificationPosture>
                {
                    Metadata = new CertificationMetadata
                    {
                        Timestamp = DateTime.UtcNow,
                        Version = "1.0",
                        Rules = "CDS-CSO-PUB",
                        Classification = "Controlled Unclassified Information"
                    },
                    Data = new List<CertificationPosture> { posture }
                };

                return Ok(result);
            }
            catch (Exception ex)
            {
                _logger.LogError(ex, "Error retrieving posture summary");
                return StatusCode(500, new { error = "Internal server error" });
            }
        }
    }

    // Data models for CDS responses

    public class CertificationDataResponse<T>
    {
        public CertificationMetadata Metadata { get; set; }
        public List<T> Data { get; set; }
    }

    public class CertificationMetadata
    {
        public DateTime Timestamp { get; set; }
        public string Version { get; set; }
        public string Rules { get; set; }
        public string Classification { get; set; }
        public Dictionary<string, object>? AdditionalInfo { get; set; }
    }

    public class SecurityDecisionEntry
    {
        public string RuleOrKsiId { get; set; }      // for example VDR-CSO-DET or KSI-IAM-APM
        public string HowFollowed { get; set; }
        public string Verification { get; set; }
        public string Validation { get; set; }
        public DateTime LastUpdated { get; set; }
        public List<string> EvidenceArtifacts { get; set; }
    }

    // Fields follow VER-RPT-VDT
    public class VulnerabilityData
    {
        public string TrackingId { get; set; }
        public string CveId { get; set; }
        public string Title { get; set; }
        public DateTime DetectedAt { get; set; }
        public string DetectionSource { get; set; }
        public DateTime EvaluatedAt { get; set; }
        public bool InternetReachable { get; set; }
        public bool LikelyExploitable { get; set; }
        public int PotentialAgencyImpactNRating { get; set; }
        public List<string> AffectedResources { get; set; }
        public bool IsOverdue { get; set; }
        public bool IsAccepted { get; set; }          // VER-TFR-MAV
        public bool IsKev { get; set; }               // VDR-TFR-KEV
        public string Disposition { get; set; }
    }

    public class VulnerabilityScanResult
    {
        public CertificationMetadata Metadata { get; set; }
        public List<VulnerabilityData> Vulnerabilities { get; set; }
    }

    public class MonitoringMetric
    {
        public DateTimeOffset Timestamp { get; set; }
        public string MetricName { get; set; }
        public int Value { get; set; }
        public Dictionary<string, object> AdditionalData { get; set; }
    }

    public class CertificationPosture
    {
        public DateTime LastUpdated { get; set; }
        public int KsiMeasuresReported { get; set; }
        public int KsiTotal { get; set; }
        public int VulnerabilitiesOpen { get; set; }
        public int VulnerabilitiesAccepted { get; set; }
        public int VulnerabilitiesOverdue { get; set; }
        public int ReportableIncidentsOpen { get; set; }
        public Dictionary<string, decimal> RuleCoverage { get; set; }
    }
}
