"""
VDR and VER Vulnerability Tracking and Reporting Module

This module sketches automated vulnerability tracking, timeframe monitoring and
reporting under the FedRAMP Consolidated Rules for 2026 (CR26): the VDR family
(Vulnerability Detection and Response) and the VER family (Vulnerability
Evaluation and Reporting). VDR and VER are required from 2026-12-07 (mandated by
CISA BOD 26-04).

Rules Implemented:
- VDR-CSO-DET: Vulnerability detection
- VDR-CSO-RES: Vulnerability response (track, evaluate, mitigate, remediate, report)
- VER-EVA-ELX, VER-EVA-EIR, VER-EVA-EPA: Evaluate exploitability, internet-reachability,
  and Potential Agency Impact N-rating
- VDR-TFR-PVR: Mitigation and remediation expectations (days from evaluation, by
  Certification Class, N-rating, internet-reachability and likely exploitability)
- VDR-TFR-KEV: Known Exploited Vulnerabilities per CISA due dates (CISA BOD 26-04)
- VER-TFR-MAV: A vulnerability not fully mitigated or remediated within 192 days of
  evaluation is an accepted vulnerability
- VER-RPT-VDT, VER-RPT-AVI, VER-RPT-PER: Report fields and persistent reporting
- VER-TFR-MHR: Report at least monthly

Related rules and KSIs:
- MAS-CSO-IIR: Scope of information resources
- KSI-PIY-GIV: Generating inventories
- KSI-SCR-MON: Monitoring supply chain risk

The VDR-TFR-PVR table below carries the values for each Certification Class from the
rules data (version 2026.10.13.01). These are SHOULD-level expectations; re-check the
rule when the rules data changes. N1 has no timeframe in the rule.

Usage:
    python frr_vdr_tracker.py --scan-results /path/to/scan.json
    python frr_vdr_tracker.py --generate-report monthly
    python frr_vdr_tracker.py --check-remediation-deadlines
"""

# Days from vulnerability evaluation, keyed by Certification Class -> N-rating -> bucket.
# Buckets: "irv_lev" internet-reachable + likely exploitable, "nirv_lev" not internet-reachable
# + likely exploitable, "nlev" not likely exploitable. Source: VDR-TFR-PVR.
# Class D, N5 uses hours/days in the rule (12 hours, 1 day, 8 days), expressed here as timedelta.
PVR_TIMEFRAMES: Dict[str, Dict[int, Dict[str, timedelta]]] = {
    "A": {
        2: {"irv_lev": timedelta(days=96), "nirv_lev": timedelta(days=160), "nlev": timedelta(days=192)},
        3: {"irv_lev": timedelta(days=32), "nirv_lev": timedelta(days=64), "nlev": timedelta(days=192)},
        4: {"irv_lev": timedelta(days=8), "nirv_lev": timedelta(days=32), "nlev": timedelta(days=64)},
        5: {"irv_lev": timedelta(days=4), "nirv_lev": timedelta(days=8), "nlev": timedelta(days=32)},
    },
    "C": {
        2: {"irv_lev": timedelta(days=48), "nirv_lev": timedelta(days=128), "nlev": timedelta(days=192)},
        3: {"irv_lev": timedelta(days=16), "nirv_lev": timedelta(days=32), "nlev": timedelta(days=128)},
        4: {"irv_lev": timedelta(days=4), "nirv_lev": timedelta(days=8), "nlev": timedelta(days=64)},
        5: {"irv_lev": timedelta(days=2), "nirv_lev": timedelta(days=4), "nlev": timedelta(days=16)},
    },
    "D": {
        2: {"irv_lev": timedelta(days=24), "nirv_lev": timedelta(days=96), "nlev": timedelta(days=192)},
        3: {"irv_lev": timedelta(days=8), "nirv_lev": timedelta(days=16), "nlev": timedelta(days=64)},
        4: {"irv_lev": timedelta(days=2), "nirv_lev": timedelta(days=8), "nlev": timedelta(days=32)},
        5: {"irv_lev": timedelta(hours=12), "nirv_lev": timedelta(days=1), "nlev": timedelta(days=8)},
    },
}
PVR_TIMEFRAMES["B"] = PVR_TIMEFRAMES["A"]  # Class B values match Class A in the rule

# VER-TFR-MAV: accepted vulnerability threshold, days from evaluation
ACCEPTED_VULNERABILITY_DAYS = 192


class RemediationStatus(Enum):
    """Response status for tracking (VDR-CSO-RES)"""
    OPEN = "Open"
    PARTIALLY_MITIGATED = "Partially Mitigated"
    FULLY_MITIGATED = "Fully Mitigated"
    REMEDIATED = "Remediated"
    ACCEPTED = "Accepted"
    FALSE_POSITIVE = "False Positive"


@dataclass
class Vulnerability:
    """Vulnerability record with fields following VER-RPT-VDT"""
    id: str  # provider's internal tracking identifier
    cve_id: Optional[str]
    title: str
    description: str
    detection_source: str
    affected_resources: List[str]
    detected_at: datetime  # time of detection
    evaluated_at: datetime  # time of completed evaluation (VER-EVA-*)
    internet_reachable: bool  # VER-EVA-EIR
    likely_exploitable: bool  # VER-EVA-ELX
    n_rating: int  # Potential Agency Impact N-rating, 1 to 5 (VER-EVA-EPA)
    target_date: Optional[datetime]  # from VDR-TFR-PVR; None for N1
    status: RemediationStatus
    is_kev: bool  # VDR-TFR-KEV flag
    accepted_reason: Optional[str] = None  # VER-RPT-AVI explanation
    last_updated: datetime = None

    def __post_init__(self):
        if self.last_updated is None:
            self.last_updated = datetime.utcnow()


class VDRComplianceTracker:
    """
    Vulnerability response tracker for VDR and VER (CR26).

    Tracks mitigation and remediation targets, accepted vulnerabilities
    and reporting inputs.
    """

    def __init__(
        self,
        storage_account_name: str,
        workspace_id: str,
        key_vault_uri: str,
        certification_class: str
    ):
        """
        Initialize the tracker

        Args:
            storage_account_name: Azure Storage account for evidence storage
            workspace_id: Log Analytics workspace ID for query operations
            key_vault_uri: Key Vault URI for credential management
            certification_class: FedRAMP Certification Class, "A", "B", "C" or "D"
        """
        if certification_class not in PVR_TIMEFRAMES:
            raise ValueError("certification_class must be one of A, B, C, D")

        self.credential = DefaultAzureCredential()
        self.storage_account_name = storage_account_name
        self.workspace_id = workspace_id
        self.certification_class = certification_class

        self.blob_client = BlobServiceClient(
            account_url=f"https://{storage_account_name}.blob.core.windows.net",
            credential=self.credential
        )
        self.logs_client = LogsQueryClient(credential=self.credential)
        self.kv_client = SecretClient(vault_url=key_vault_uri, credential=self.credential)

        logger.info("VDR/VER tracker initialized for Class %s", certification_class)

    def calculate_target_date(
        self,
        evaluated_at: datetime,
        n_rating: int,
        internet_reachable: bool,
        likely_exploitable: bool
    ) -> Optional[datetime]:
        """
        Calculate the mitigation/remediation target per VDR-TFR-PVR

        Timeframes depend on the Certification Class, the Potential Agency Impact
        N-rating, internet-reachability and likely exploitability. There are no fixed
        severity buckets in CR26. Returns None for N1 (no timeframe in the rule).
        """
        table = PVR_TIMEFRAMES[self.certification_class].get(n_rating)
        if not table:
            return None

        if not likely_exploitable:
            bucket = "nlev"
        elif internet_reachable:
            bucket = "irv_lev"
        else:
            bucket = "nirv_lev"

        target = evaluated_at + table[bucket]
        logger.debug("Target %s for N%d bucket %s", target, n_rating, bucket)
        return target

    def accepted_threshold(self, evaluated_at: datetime) -> datetime:
        """VER-TFR-MAV: not fully mitigated or remediated by this date means accepted."""
        return evaluated_at + timedelta(days=ACCEPTED_VULNERABILITY_DAYS)

    def ingest_scan_results(self, scan_results_path: str) -> List[Vulnerability]:
        """
        Ingest detection results (VDR-CSO-DET) and evaluate them (VER-EVA-*)

        Supports multiple formats (Trivy JSON shown; Qualys XML, Tenable CSV and
        Defender JSON can be converted to JSON first).

        The N-rating, reachability and exploitability below are placeholders: real
        evaluation needs your architecture context (VER-EVA-EIR, VER-EVA-ELX, VER-EVA-EPA).
        """
        logger.info("Ingesting scan results from %s", scan_results_path)

        with open(scan_results_path, 'r') as f:
            scan_data = json.load(f)

        vulnerabilities = []
        now = datetime.utcnow()

        for result in scan_data.get('Results', []):
            for vuln in result.get('Vulnerabilities', []):
                cve_id = vuln.get('VulnerabilityID')

                # Placeholder evaluation. Replace with your evaluation process.
                internet_reachable = False
                likely_exploitable = True  # VER-EVA-AIA: assume exploitation can be automated unless shown otherwise
                n_rating = 3

                vulnerability = Vulnerability(
                    id=f"{cve_id}_{result.get('Target', 'unknown')}",
                    cve_id=cve_id,
                    title=vuln.get('Title', 'Unknown'),
                    description=vuln.get('Description', ''),
                    detection_source="trivy",
                    affected_resources=[result.get('Target', 'unknown')],
                    detected_at=now,
                    evaluated_at=now,
                    internet_reachable=internet_reachable,
                    likely_exploitable=likely_exploitable,
                    n_rating=n_rating,
                    target_date=self.calculate_target_date(
                        now, n_rating, internet_reachable, likely_exploitable),
                    status=RemediationStatus.OPEN,
                    is_kev=self.check_kev_status(cve_id)
                )

                vulnerabilities.append(vulnerability)

        logger.info("Ingested %d vulnerabilities", len(vulnerabilities))
        return vulnerabilities

    def store_vulnerabilities(
        self,
        vulnerabilities: List[Vulnerability],
        container_name: str = "vulnerability-scans"
    ) -> None:
        """
        Store vulnerability records as JSON (VER-RPT-PER, VER-RPT-VDT)

        Persistent vulnerability reports are FedRAMP Certification Data.
        """
        container_client = self.blob_client.get_container_client(container_name)

        timestamp = datetime.utcnow().strftime("%Y-%m-%d-%H%M%S")
        blob_name = f"{timestamp}-vulnerability-report.json"

        def serialize(v: Vulnerability) -> Dict:
            record = asdict(v)
            record["status"] = v.status.value
            for key in ("detected_at", "evaluated_at", "target_date", "last_updated"):
                value = getattr(v, key)
                record[key] = value.isoformat() if value else None
            return record

        data = {
            "metadata": {
                "timestamp": datetime.utcnow().isoformat(),
                "version": "1.0",
                "rules": ["VDR-CSO-DET", "VER-RPT-PER", "VER-RPT-VDT", "VER-TFR-MHR"],
                "certification_class": self.certification_class,
                "total_vulnerabilities": len(vulnerabilities),
                "n4_or_n5_count": sum(1 for v in vulnerabilities if v.n_rating >= 4),
                "kev_count": sum(1 for v in vulnerabilities if v.is_kev)
            },
            "vulnerabilities": [serialize(v) for v in vulnerabilities]
        }

        blob_client = container_client.get_blob_client(blob_name)
        blob_client.upload_blob(json.dumps(data, indent=2))

        logger.info("Stored vulnerabilities in %s", blob_name)

    def check_overdue_remediation(self) -> List[Dict]:
        """
        Find vulnerabilities past their VDR-TFR-PVR target or past the
        192-day accepted threshold (VER-TFR-MAV).

        Returns:
            List of overdue vulnerabilities with details
        """
        logger.info("Checking for overdue vulnerabilities")

        container_client = self.blob_client.get_container_client("vulnerability-scans")

        overdue = []
        now = datetime.utcnow()

        blobs = list(container_client.list_blobs())
        if not blobs:
            return overdue
        latest_blob = sorted(blobs, key=lambda b: b.name, reverse=True)[0]

        blob_client = container_client.get_blob_client(latest_blob.name)
        report = json.loads(blob_client.download_blob().readall())

        for vuln in report.get('vulnerabilities', []):
            if vuln['status'] in (RemediationStatus.REMEDIATED.value, RemediationStatus.ACCEPTED.value,
                                  RemediationStatus.FALSE_POSITIVE.value):
                continue

            evaluated_at = datetime.fromisoformat(vuln['evaluated_at'])
            target = datetime.fromisoformat(vuln['target_date']) if vuln.get('target_date') else None
            past_target = target is not None and target < now
            past_accept = self.accepted_threshold(evaluated_at) < now

            if past_target or past_accept:
                overdue.append({
                    "id": vuln['id'],
                    "cve_id": vuln['cve_id'],
                    "n_rating": vuln['n_rating'],
                    "target_date": vuln.get('target_date'),
                    "past_pvr_target": past_target,
                    "must_mark_accepted": past_accept,  # VER-TFR-MAV, report per VER-RPT-AVI
                    "affected_resources": vuln['affected_resources']
                })

        logger.info("Found %d overdue vulnerabilities", len(overdue))
        return overdue

    def generate_monthly_report(self) -> Dict:
        """
        Generate the periodic vulnerability report (VER-TFR-MHR: at least monthly)

        Reports summarize ALL activity since the previous report (VER-RPT-PER) and

        """

            "summary": {
                "total_kev": 0,

def main():
    """Main execution function"""

    storage_account = os.getenv('EVIDENCE_STORAGE_ACCOUNT', 'vdrevidence')
    workspace_id = os.getenv('LOG_ANALYTICS_WORKSPACE_ID')
    key_vault_uri = os.getenv('KEY_VAULT_URI')
    certification_class = os.getenv('FEDRAMP_CERTIFICATION_CLASS', 'B')

    tracker = VDRComplianceTracker(
        storage_account_name=storage_account,
        workspace_id=workspace_id,
        key_vault_uri=key_vault_uri,
        certification_class=certification_class
    )

    scan_results_path = os.getenv('SCAN_RESULTS_PATH', './scan-results.json')
    if os.path.exists(scan_results_path):
        vulnerabilities = tracker.ingest_scan_results(scan_results_path)
        tracker.store_vulnerabilities(vulnerabilities)

    overdue = tracker.check_overdue_remediation()
    if overdue:
        logger.warning("%d vulnerabilities are past a VDR/VER threshold", len(overdue))
        # Send alert

    tracker.generate_monthly_report()
    logger.info("Report generated")


if __name__ == "__main__":
    main()
